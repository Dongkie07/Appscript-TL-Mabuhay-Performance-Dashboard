const selectedLevel = getElementById("slotLevelFilter").value; const selectedSort = getElementById("slotSort").value; const filtered = mergedBranches.filter(function filterBoth(merged) { const matchesSearch = !searchText || merged.branchName.toLowerCase().includes(searchText) || merged.region.toLowerCase().includes(searchText); const status = branch.trend.status || "INSUFFICIENT_DATA"; ranks.push( Object.prototype.hasOwnProperty.call(trendOrder, status) ? trendOrder[status] : trendOrder.INSUFFICIENT_DATA, ); }); return ranks.length ?
